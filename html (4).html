/* 定数 */
// ★IndexedDB対応のため、この制限はブラウザの物理的制限ではなく、あくまで表示上の目安になります。
const MAX_STORAGE_BYTES = 1048576 * 300; 
const DB_NAME = "WebWinFiles";
const DB_VERSION = 1;
const STORE_NAME = "files";

/* IndexedDBの初期化と取得 */
let dbPromise;

function getDbPromise() {
    if (!dbPromise) {
        dbPromise = new Promise((resolve, reject) => {
            const request = indexedDB.open(DB_NAME, DB_VERSION);

            request.onerror = (event) => {
                console.error("IndexedDBエラー:", event.target.errorCode);
                alert("データベースを開けませんでした。");
                reject(event.target.error);
            };

            request.onsuccess = (event) => {
                resolve(event.target.result);
            };

            request.onupgradeneeded = (event) => {
                const db = event.target.result;
                if (!db.objectStoreNames.contains(STORE_NAME)) {
                    db.createObjectStore(STORE_NAME, { keyPath: "filename" });
                }
            };
        });
    }
    return dbPromise;
}

// IndexedDBへのトランザクションを実行するヘルパー
async function transaction(mode) {
    const db = await getDbPromise();
    const tx = db.transaction(STORE_NAME, mode);
    const store = tx.objectStore(STORE_NAME);
    return { store, tx };
}

/* ファイル操作 - IndexedDB対応 (すべて非同期/awaitが必要) */

/**
 * ファイルをIndexedDBに保存する
 * @param {string} filename - ファイル名 (キー)
 * @param {string} content - ファイルの中身 (Data URL)
 * @returns {Promise<boolean>} 成功したらtrue
 */
async function saveFileToIndexedDB(filename, content) {
    try {
        const { store, tx } = await transaction("readwrite");
        const fileRecord = { filename, content, size: content.length * 2 }; // サイズは概算
        
        await new Promise((resolve, reject) => {
            const request = store.put(fileRecord);
            request.onsuccess = () => resolve();
            request.onerror = (e) => reject(e.target.error);
        });

        await new Promise((resolve, reject) => {
            tx.oncomplete = () => resolve(true);
            tx.onerror = (e) => reject(e.target.error);
            tx.onabort = () => reject(new Error('Transaction aborted'));
        });
        return true;
    } catch (e) {
        console.error("ファイル保存エラー:", e);
        if (e.name === 'QuotaExceededError') {
             alert(`警告！保存容量制限を超えました。\nブラウザの設定で保存できませんでした。`);
        } else {
             alert(`ファイル保存中にエラーが発生しました: ${e.message}`);
        }
        return false;
    }
}

/**
 * ファイルをIndexedDBから読み込む
 * @param {string} filename - ファイル名 (キー)
 * @returns {Promise<{filename: string, content: string, size: number} | null>} ファイルレコード
 */
async function getFileFromIndexedDB(filename) {
    try {
        const { store } = await transaction("readonly");
        return await new Promise((resolve, reject) => {
            const request = store.get(filename);
            request.onsuccess = () => resolve(request.result);
            request.onerror = (e) => reject(e.target.error);
        });
    } catch (e) {
        console.error("ファイル読み込みエラー:", e);
        return null;
    }
}

/**
 * ファイルをIndexedDBから削除する
 * @param {string} filename - ファイル名 (キー)
 * @returns {Promise<boolean>} 成功したらtrue
 */
async function deleteFileFromIndexedDB(filename) {
    try {
        const { store, tx } = await transaction("readwrite");
        await new Promise((resolve, reject) => {
            const request = store.delete(filename);
            request.onsuccess = () => resolve();
            request.onerror = (e) => reject(e.target.error);
        });
        await new Promise(resolve => { tx.oncomplete = resolve; }); // トランザクション完了を待つ
        return true;
    } catch (e) {
        console.error("ファイル削除エラー:", e);
        return false;
    }
}

/**
 * すべてのファイル名と概算サイズを取得する
 * @returns {Promise<Array<{filename: string, size: number}>>}
 */
async function listAllFiles() {
    try {
        const { store } = await transaction("readonly");
        return await new Promise((resolve, reject) => {
            const files = [];
            const request = store.openCursor();
            request.onsuccess = (event) => {
                const cursor = event.target.result;
                if (cursor) {
                    files.push({ filename: cursor.value.filename, size: cursor.value.size || 0 });
                    cursor.continue();
                } else {
                    resolve(files);
                }
            };
            request.onerror = (e) => reject(e.target.error);
        });
    } catch (e) {
        console.error("ファイル一覧取得エラー:", e);
        return [];
    }
}

/**
 * 全データを削除する（フォーマット）関数
 */
async function formatStorage(){
    const allFiles = await listAllFiles();
    if(allFiles.length === 0){
        alert("削除するファイルがありませんでした。");
        return;
    }

    if(confirm(`警告: すべてのデータ (${allFiles.length}個のファイル) を完全に削除します。よろしいですか？\n(この操作は元に戻せません)`)){
        try {
            const { store, tx } = await transaction("readwrite");
            await new Promise((resolve, reject) => {
                const request = store.clear();
                request.onsuccess = () => resolve();
                request.onerror = (e) => reject(e.target.error);
            });
            await new Promise(resolve => { tx.oncomplete = resolve; });

            alert("ストレージをフォーマットし、すべてのファイルを削除しました。");
            updateExplorerViews(); 
        } catch(e) {
            alert(`フォーマット中にエラーが発生しました: ${e.message}`);
        }
    }
}


/* 使用量概算 (IndexedDB対応) */
async function getUsedStorageBytes(){
    const files = await listAllFiles();
    let total = 0;
    files.forEach(f => { total += f.size; });
    
    // ブラウザのQuota APIがあれば使う（今回はシンプルに概算のみで）
    return total;
}

let zIndexCounter = 2000;
let winIdCounter = 0;
const winRegistry = {}; 

function createWindow({title='Window', contentHTML='', width=700, height=420, icon=null}){
    const id = 'win_' + (++winIdCounter);
    const win = document.createElement('div');
    win.className = 'window';
    win.style.width = width + 'px';
    win.style.height = height + 'px';
    const offset = (winIdCounter % 6) * 18;
    const cx = Math.max(12, (innerWidth - width)/2 + offset);
    const cy = Math.max(24, (innerHeight - height - 40)/2 + offset);
    win.style.left = cx + 'px';
    win.style.top = cy + 'px';
    win.style.zIndex = ++zIndexCounter;
    win.dataset.winId = id;
    win.title = title;

    win.innerHTML = `
        <div class="title-bar">
            <div class="title-left"><strong>${title}</strong></div>
            <div class="title-btns">
                <button class="minimize" title="最小化">—</button>
                <button class="maximize" title="最大化">▢</button>
                <button class="close" title="閉じる">✕</button>
            </div>
        </div>
        <div class="content">${contentHTML}</div>
    `;

    document.body.appendChild(win);
    winRegistry[id] = { win, title, icon };

    win.addEventListener('mousedown', () => { win.style.zIndex = ++zIndexCounter; updateTaskbar(); });
    win.addEventListener('touchstart', () => { win.style.zIndex = ++zIndexCounter; updateTaskbar(); });

    const titleBar = win.querySelector('.title-bar');
    let dragging=false, sx=0, sy=0, ox=0, oy=0;
    
    function getCoords(e){
        if(e.touches && e.touches.length) return { clientX: e.touches[0].clientX, clientY: e.touches[0].clientY };
        return { clientX: e.clientX, clientY: e.clientY };
    }

    function dragStart(e) {
        if(e.target.closest('.title-btns')) return; 
        e.preventDefault(); 
        const coords = getCoords(e);
        dragging=true; 
        sx=coords.clientX; 
        sy=coords.clientY; 
        ox=parseFloat(win.style.left)||0; 
        oy=parseFloat(win.style.top)||0; 
        win.style.transition='none';
    }
    
    function dragMove(e) { 
        if(!dragging) return; 
        const coords = getCoords(e);
        win.style.left = (ox + (coords.clientX - sx)) + 'px'; 
        win.style.top = (oy + (coords.clientY - sy)) + 'px'; 
    }
    
    function dragEnd() { 
        if(dragging){
            dragging=false; 
            win.style.transition='box-shadow 0.1s, transform 0.2s';
        } 
    }
    
    titleBar.addEventListener('mousedown', dragStart);
    document.addEventListener('mousemove', dragMove);
    document.addEventListener('mouseup', dragEnd);
    
    titleBar.addEventListener('touchstart', dragStart);
    document.addEventListener('touchmove', dragMove);
    document.addEventListener('touchend', dragEnd);

    const btnMin = win.querySelector('.minimize');
    const btnMax = win.querySelector('.maximize');
    const btnClose = win.querySelector('.close');
    
    [btnMin, btnMax, btnClose].forEach(btn => {
        btn.addEventListener('click', e => { e.stopPropagation(); });
        btn.addEventListener('touchend', e => { e.stopPropagation(); });
    });

    let prevRect = null;
    btnMin.onclick = () => { win.style.display = 'none'; updateTaskbar(); };
    btnClose.onclick = () => { 
        // 音楽プレーヤーを閉じる際に再生を停止
        if(win.dataset.app === 'media-player') {
            const audioEl = win.querySelector('audio');
            if(audioEl) audioEl.pause();
        }
        win.remove(); delete winRegistry[id]; updateTaskbar(); updateExplorerViews(); 
    }; 
    btnMax.onclick = () => {
        if(!prevRect){
            prevRect = {left:win.style.left, top:win.style.top, width:win.style.width, height:win.style.height};
            win.style.left = '0px'; win.style.top = '0px'; win.style.width = '100vw'; win.style.height = 'calc(100vh - var(--taskbar-h))';
            btnMax.textContent = '❐'; 
        } else {
            win.style.left = prevRect.left; win.style.top = prevRect.top; win.style.width = prevRect.width; win.style.height = prevRect.height; prevRect = null;
            btnMax.textContent = '▢'; 
        }
    };

    registerWindowToTaskbar(id, title, icon);

    return { id, win };
}

const taskbarAppsEl = document.getElementById('taskbar-apps');

function registerWindowToTaskbar(id, title, iconUrl){
    if(taskbarAppsEl.querySelector(`[data-win="${id}"]`)) return;
    const el = document.createElement('div');
    el.className = 'taskbar-icon';
    el.dataset.win = id;
    const img = document.createElement('img');
    img.alt = title;
    img.draggable = false;
    img.src = iconUrl || 'https://img.icons8.com/fluency/48/windows-10.png';
    el.appendChild(img);
    el.title = title;
    el.onclick = () => {
        const obj = winRegistry[id];
        if(!obj){ el.remove(); updateTaskbar(); return; }
        const w = obj.win;
        if(w.style.display === 'none' || window.getComputedStyle(w).display === 'none'){
            w.style.display = 'flex';
            w.style.zIndex = ++zIndexCounter;
        } else {
            w.style.display = 'none';
        }
        updateTaskbar();
    };
    taskbarAppsEl.appendChild(el);
    updateTaskbar();
}

function updateTaskbar(){
    const icons = taskbarAppsEl.querySelectorAll('.taskbar-icon');
    icons.forEach(ic => {
        const id = ic.dataset.win;
        const reg = winRegistry[id];
        if(!reg){ ic.remove(); return; }
        const visible = reg.win.style.display !== 'none' && window.getComputedStyle(reg.win).display !== 'none';
        ic.classList.toggle('active', visible);
    });
}

/* Webブラウザ (変更なし) */
function launchBrowser(){
    for(const id in winRegistry){ if(winRegistry[id].title === 'Webブラウザ'){ const obj = winRegistry[id]; obj.win.style.display='flex'; obj.win.style.zIndex = ++zIndexCounter; updateTaskbar(); return; } }
    
    const localId = winIdCounter + 1;
    const browserHTML = `
        <div style="display:flex;flex-direction:column;height:100%;">
            <div class="tab-container-bar"><div class="tab-bar" id="tabbar_${localId}"></div><button id="newtab_${localId}" style="margin-left:auto;padding:4px 8px;">＋ 新しいタブ</button></div>
            <div style="display:flex;gap:6px;padding:6px;background:#f0f0f0;border-bottom:1px solid #ccc;flex-shrink:0;">
                <input id="url_${localId}" type="text" style="flex:1;padding:6px;border:1px solid #ccc;border-radius:4px;" value="https://example.com" placeholder="URLを入力してEnter">
                <button id="gobtn_${localId}">開く</button>
            </div>
            <div id="iframes_${localId}" style="flex:1;overflow:hidden;min-height:0; position: relative;"></div>
        </div>
    `;
    const { id, win } = createWindow({title:'Webブラウザ', contentHTML:browserHTML, width:840, height:560, icon:'https://img.icons8.com/color/48/internet--v1.png'});
    
    const tabbar = win.querySelector(`#tabbar_${localId}`);
    const iframes = win.querySelector(`#iframes_${localId}`);
    const urlInput = win.querySelector(`#url_${localId}`);
    const goBtn = win.querySelector(`#gobtn_${localId}`);
    const newTabBtn = win.querySelector(`#newtab_${localId}`);
    let tabs = []; 
    let tabIndex = 0;
    let activeTabIdx = -1;

    function createTab(url, activate=true){
        const idx = tabIndex++;
        const tabEl = document.createElement('div'); 
        tabEl.className='tab'; 
        tabEl.innerHTML = `<span>${(url.replace(/^https?:\/\//,'').split('/')[0]||'新しいタブ')}</span> <span class="close-tab-btn" data-idx="${idx}" style="font-weight:bold;margin-left:4px;">✕</span>`;
        tabEl.dataset.idx = idx;
        
        const frame = document.createElement('iframe'); 
        frame.src = url;
        frame.sandbox = "allow-same-origin allow-scripts allow-forms allow-downloads allow-modals allow-top-navigation-by-user"; 
        
        const tabWrap = document.createElement('div'); 
        tabWrap.className = 'tab-content'; 
        tabWrap.dataset.idx = idx;
        tabWrap.appendChild(frame);

        tabbar.appendChild(tabEl);
        iframes.appendChild(tabWrap);
        
        tabEl.querySelector('span').onclick = ()=> setActive(idx); 
        tabEl.querySelector('.close-tab-btn').addEventListener('click', (e) => closeTab(e, idx));

        tabs.push({idx, tabEl, tabWrap, url});
        if(activate) setActive(idx);
    }

    function closeTab(e, idx){
        e.stopPropagation(); 
        const tab = tabs.find(t=>t.idx===idx);
        if(!tab) return;
        
        tab.tabEl.remove();
        tab.tabWrap.remove();
        tabs = tabs.filter(t=>t.idx!==idx);
        
        if(tabs.length === 0){
            win.remove(); delete winRegistry[id]; updateTaskbar(); return;
        }
        
        if(activeTabIdx === idx){
            setActive(tabs[tabs.length-1].idx); 
        }
    }

    function setActive(idx){
        activeTabIdx = idx;
        tabs.forEach(t=>{ 
            const active = t.idx === idx; 
            t.tabEl.classList.toggle('active', active); 
            t.tabWrap.classList.toggle('active', active); 
            if(active) urlInput.value = t.url; 
        });
    }

    function navigate(){
        let url = urlInput.value;
        if(!url.startsWith('http')) url = 'https://' + url;
        const active = tabs.find(t=>t.idx === activeTabIdx);
        if(active){ 
            active.url = url; 
            active.tabWrap.querySelector('iframe').src = url; 
            active.tabEl.querySelector('span').textContent = url.replace(/^https?:\/\//,'').split('/')[0]; 
        }
    }

    newTabBtn.onclick = ()=> createTab('https://example.com', true);
    goBtn.onclick = navigate;
    urlInput.addEventListener('keydown', (e) => { if(e.key === 'Enter') navigate(); });
    
    createTab('https://example.com', true);
}


/* メディアビューア (画像/動画) - IndexedDB対応 (awaitでファイル内容を取得) */
async function openMediaViewer(filename){
    const fileRecord = await getFileFromIndexedDB(filename);
    if (!fileRecord) { alert(`ファイル「${filename}」が見つかりませんでした。`); return; }
    const dataUrl = fileRecord.content;
    
    for(const id in winRegistry){
        if(winRegistry[id].title === filename){
            const obj = winRegistry[id]; obj.win.style.display='flex'; obj.win.style.zIndex = ++zIndexCounter; updateTaskbar(); return;
        }
    }

    let mediaElement = '';
    const isImage = dataUrl.startsWith('data:image/');
    const isVideo = dataUrl.startsWith('data:video/');

    if (isImage) {
        mediaElement = `<img src="${dataUrl}" alt="${filename}">`;
    } else if (isVideo) {
        mediaElement = `<video controls autoplay src="${dataUrl}" type="${dataUrl.split(';')[0].replace('data:','')}" style="max-height:100%; max-width:100%;"></video>`;
    } else {
        alert(`「${filename}」は画像・動画ファイルとして認識できません。`);
        return;
    }

    const html = `
        <div class="media-viewer-content">
            ${mediaElement}
        </div>
    `;

    const { id, win } = createWindow({title: filename, contentHTML:html, width:800, height:600, icon: isImage ? 'https://img.icons8.com/fluency/48/picture.png' : 'https://img.icons8.com/fluency/48/video.png'});
    win.style.backgroundColor = '#000';
    updateTaskbar();
}

/* 音楽プレーヤー - IndexedDB対応 (awaitでファイル内容を取得) */
async function openMediaPlayer(filename){
    const fileRecord = await getFileFromIndexedDB(filename);
    if (!fileRecord) { alert(`ファイル「${filename}」が見つかりませんでした。`); return; }
    const dataUrl = fileRecord.content;

    if (!dataUrl.startsWith('data:audio/')) {
        alert(`ファイル「${filename}」は音声ファイルとして認識できません。`);
        return;
    }

    for(const id in winRegistry){
        if(winRegistry[id].title === filename){
            const obj = winRegistry[id]; obj.win.style.display='flex'; obj.win.style.zIndex = ++zIndexCounter; updateTaskbar(); return;
        }
    }

    const html = `
        <div class="music-player-content">
            <img src="https://img.icons8.com/fluency/96/music.png" alt="music icon">
            <h3>${filename}</h3>
            <audio controls autoplay loop>
                <source src="${dataUrl}" type="${dataUrl.split(';')[0].replace('data:','')}">
                お使いのブラウザは音声ファイルをサポートしていません。

        </div>
    `;
    
    const { id, win } = createWindow({title: filename, contentHTML:html, width:500, height:280, icon:'https://img.icons8.com/fluency/48/music.png'});
    win.dataset.app = 'media-player'; 
    updateTaskbar();
}


/* エクスプローラー - IndexedDB対応 (非同期処理に書き換え) */
function openExplorer(){
    for(const id in winRegistry){ 
        if(winRegistry[id].title === 'エクスプローラー'){ 
            const obj = winRegistry[id]; 
            obj.win.style.display='flex'; 
            obj.win.style.zIndex=++zIndexCounter; 
            if(typeof obj.win.renderFiles === 'function') obj.win.renderFiles(); 
            updateTaskbar(); 
            return; 
        } 
    }
    
    const localId = winIdCounter + 1;
    const html = `
        <div style="display:flex;flex-direction:column;height:100%;">
            <div style="display:flex;align-items:center;gap:8px;padding:6px;background:#f0f0f0;border-bottom:1px solid #ccc;flex-shrink:0;">
                <strong>📁 ドキュメント</strong>
                <button id="newtxt_${localId}">新しいメモ</button>
                <input type="file" id="fileinput_${localId}" style="display:none;" multiple accept="*/*">
                <button id="uploadbtn_${localId}">⬆️ ファイルをアップロード</button>
                <button id="formatbtn_${localId}" style="margin-left:auto; background:#f00; color:white; border:none; padding:6px 10px; border-radius:4px;">⚠️ すべて削除 (初期化)</button>
            </div>
            <div class="content-view" id="filearea_${localId}" style="flex:1;overflow:auto;padding:10px;"></div>
            <div style="padding:6px;background:#f0f0f0;border-top:1px solid #ccc;font-size:13px;flex-shrink:0;" id="status_${localId}"></div>
        </div>
    `;
    const { id, win } = createWindow({title:'エクスプローラー', contentHTML:html, width:660, height:460, icon:'https://img.icons8.com/fluency/48/folder-invoices.png'});
    const area = win.querySelector(`#filearea_${localId}`);
    const status = win.querySelector(`#status_${localId}`);
    const newBtn = win.querySelector(`#newtxt_${localId}`);
    const uploadBtn = win.querySelector(`#uploadbtn_${localId}`); 
    const fileInput = win.querySelector(`#fileinput_${localId}`); 
    const formatBtn = win.querySelector(`#formatbtn_${localId}`); 

    async function updateStatus(){
        const used = await getUsedStorageBytes();
        const usedKB = Math.round(used/1024);
        const usedMB = (used/1048576).toFixed(2);
        const maxMB = (MAX_STORAGE_BYTES/1048576).toFixed(0);
        status.textContent = `容量: ${usedKB}KB (${usedMB}MB) / ${maxMB}MB 概算使用中`;
        status.style.color = used >= MAX_STORAGE_BYTES ? 'red' : (used >= MAX_STORAGE_BYTES*0.9 ? 'orange' : 'black');
    }

    async function renderFiles(){
        area.innerHTML = '<div style="padding:24px;color:#666">読み込み中...</div>';
        const allFiles = await listAllFiles();
        area.innerHTML = ''; 
        let count=0;

        for(const file of allFiles){
            const fname = file.filename;
            count++;

            let icon='https://img.icons8.com/fluency/48/file.png';
            let clickAction = ()=> openNotepad(fname);

            // ファイルタイプに応じてアイコンとアクションを設定 (ファイル名で判断)
            const lowerFname = fname.toLowerCase();

            if(lowerFname.endsWith('.txt')){
                icon='https://img.icons8.com/fluency/48/text-file.png';
                clickAction = ()=> openNotepad(fname);
            } else if(lowerFname.endsWith('.png') || lowerFname.endsWith('.jpg') || lowerFname.endsWith('.gif') || lowerFname.endsWith('.webp')){
                icon='https://img.icons8.com/fluency/48/picture.png';
                // 画像ファイルの内容は表示時に取得する
                clickAction = ()=> openMediaViewer(fname);
            } else if(lowerFname.endsWith('.mp4') || lowerFname.endsWith('.webm')){
                 icon='https://img.icons8.com/fluency/48/video.png';
                 clickAction = ()=> openMediaViewer(fname);
            } else if(lowerFname.endsWith('.mp3') || lowerFname.endsWith('.wav') || lowerFname.endsWith('.ogg')){
                 icon='https://img.icons8.com/fluency/48/music.png';
                 clickAction = ()=> openMediaPlayer(fname);
            }

            // アイコンのサムネイル表示は今回はスキップ (画像の場合もopenMediaViewerに任せる)
            const iconElement = `<img src="${icon}" alt="${fname}">`;
            
            const el = document.createElement('div'); 
            el.className='file-icon';
            el.title = fname;

            el.innerHTML = `${iconElement}<div>${fname}</div><button class="file-delete-btn" data-fname="${fname}">✕</button>`;
            el.onclick = clickAction;
            
            el.querySelector('.file-delete-btn').addEventListener('click', async (e) => {
                e.stopPropagation();
                const targetFname = e.target.dataset.fname;
                if(confirm(`「${targetFname}」を削除してもいい？`)){
                    const success = await deleteFileFromIndexedDB(targetFname);
                    if(success) {
                        alert(`「${targetFname}」を削除したよ。`);
                        renderFiles(); 
                    } else {
                        alert(`「${targetFname}」の削除に失敗しました。`);
                    }
                }
            });

            area.appendChild(el);
        }
        
        if(count===0) area.innerHTML = `<div style="padding:24px;color:#666">ここにファイルはありません。</div>`;
        await updateStatus();
    }

    formatBtn.onclick = formatStorage; 
    
    uploadBtn.onclick = () => { fileInput.click(); }; 
    fileInput.onchange = async (e) => {
        if (e.target.files.length === 0) return;
        
        // ファイルの競合チェック
        const existingFiles = (await listAllFiles()).map(f => f.filename);

        for (const file of e.target.files) {
            let filename = file.name;
            let counter = 1;
            while(existingFiles.includes(filename)) { // 既存ファイル名と衝突しないようにリネーム
                const parts = file.name.split('.');
                const ext = parts.pop();
                const base = parts.join('.');
                filename = `${base}(${counter++}).${ext}`;
            }

            const reader = new FileReader();
            reader.onload = async () => {
                const dataUrl = reader.result; 
                const success = await saveFileToIndexedDB(filename, dataUrl);
                
                if(success){
                    alert(`「${filename}」をアップロードしました！`);
                    existingFiles.push(filename); // リストに加える
                    renderFiles(); 
                }
            };
            reader.onerror = () => {
                alert(`「${file.name}」の読み込み中にエラーが発生しました。`);
            };
            reader.readAsDataURL(file); 
        }
        fileInput.value = '';
    };
    
    newBtn.onclick = ()=> openNotepad();
    win.renderFiles = renderFiles; 
    renderFiles();
    updateTaskbar();
}

/* updateExplorerViews (非同期に対応) */
function updateExplorerViews(){
    document.querySelectorAll('.window').forEach(w => {
        try{ 
            if(w.title === 'エクスプローラー' && typeof w.renderFiles === 'function') w.renderFiles(); 
        } catch(e){}
    });
}

/* メモ帳 - IndexedDB対応 (非同期処理に書き換え) */
async function openNotepad(initialFilename = "無題.txt"){
    for(const id in winRegistry){
        if(winRegistry[id].title === initialFilename || (initialFilename === '無題.txt' && winRegistry[id].title === 'メモ帳')){
            const o = winRegistry[id]; o.win.style.display='flex'; o.win.style.zIndex = ++zIndexCounter; updateTaskbar(); return;
        }
    }
    
    const localId = winIdCounter + 1;
    let currentFilename = initialFilename;
    const html = `
        <div style="display:flex;flex-direction:column;height:100%;">
            <div style="display:flex;align-items:center;gap:8px;padding:6px;background:#f0f0f0;border-bottom:1px solid #ccc;flex-shrink:0;">
                <button id="savebtn_${localId}">💾 保存</button>
                <button id="delbtn_${localId}" style="margin-left:auto;background:#fdd;padding:6px;border-radius:6px;border:1px solid #f99;">🗑 削除</button>
            </div>
            <textarea class="notepad-textarea" id="notearea_${localId}" placeholder="ここにメモを書いてね..."></textarea>
        </div>
    `;
    const { id, win } = createWindow({title: currentFilename === '無題.txt' ? 'メモ帳' : currentFilename, contentHTML:html, width:520, height:360, icon:'https://img.icons8.com/fluency/48/note.png'});
    const textarea = win.querySelector(`#notearea_${localId}`);
    const savebtn = win.querySelector(`#savebtn_${localId}`);
    const delbtn = win.querySelector(`#delbtn_${localId}`);

    async function loadFile(){
        if(currentFilename === '無題.txt') return;

        const fileRecord = await getFileFromIndexedDB(currentFilename);
        if(fileRecord){
            const data = fileRecord.content;
            const isBinary = data.startsWith('data:image/') || data.startsWith('data:video/') || data.startsWith('data:audio/'); 
            if(isBinary){
                textarea.value = `[バイナリファイル: ${currentFilename}]\n\n※画像、動画、音声などのバイナリファイルは、メモ帳で内容を編集・表示することはできません。`;
                textarea.disabled = true;
                savebtn.disabled = true;
            } else {
                textarea.value = data;
                textarea.disabled = false;
                savebtn.disabled = false;
            }
        }
    }

    async function saveFile(){
        if(textarea.disabled) { alert('このファイルは編集できません。'); return; }

        let filename = prompt("ファイル名を入力してください（例: memo.txt）", currentFilename === '無題.txt' ? '' : currentFilename);
        if(!filename) return;
        if(!filename.toLowerCase().endsWith('.txt')) filename += '.txt';
        
        const newText = textarea.value || '';
        // Data URL形式に変換（テキストファイルなのでtext/plainのData URLを使う）
        const dataUrl = `data:text/plain;charset=utf-8,${encodeURIComponent(newText)}`;

        const success = await saveFileToIndexedDB(filename, dataUrl);
        
        if(success){
            if(currentFilename !== '無題.txt' && filename !== currentFilename){
                await deleteFileFromIndexedDB(currentFilename);
            }

            currentFilename = filename;
            winRegistry[id].title = currentFilename; 
            win.querySelector('.title-left strong').textContent = winRegistry[id].title;
            win.title = winRegistry[id].title;
            updateExplorerViews();
            alert(`「${filename}」を保存したよ！`);
            updateTaskbar();
        }
    }
    
    async function deleteFile(){
        if(currentFilename === '無題.txt'){ alert('無題のファイルは削除できないよ。'); return; }
        if(!confirm(`「${currentFilename}」を削除してもいい？`)) return;
        
        const success = await deleteFileFromIndexedDB(currentFilename);
        if(success) {
            alert(`「${currentFilename}」を削除したよ。`);
            win.remove();
            delete winRegistry[id]; 
            updateExplorerViews();
            updateTaskbar();
        } else {
            alert(`「${currentFilename}」の削除に失敗しました。`);
        }
    }
    savebtn.onclick = saveFile;
    delbtn.onclick = deleteFile;
    loadFile();
    updateTaskbar();
}

        if(changed) w.style.zIndex = ++zIndexCounter;
    });
});
